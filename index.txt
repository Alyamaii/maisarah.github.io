    <!-- Application Logic Scripts -->
    <script>
        // 1. Define Mathematical Equations
        // f(x) = -0.0008x^2 + 0.24x + 12
        function getYield(x) {
            return (-0.0008 * Math.pow(x, 2)) + (0.24 * x) + 12;
        }

        // f'(x) = -0.0016x + 0.24
        function getDerivative(x) {
            return (-0.0016 * x) + 0.24;
        }

        // 2. Generate Data Points for Plotting (Nitrogen 0 to 250 kg/ha)
        const xValues = [];
        const yValues = [];
        for (let x = 0; x <= 250; x += 10) {
            xValues.push(x);
            yValues.push(getYield(x).toFixed(2));
        }

        // 3. Setup ChartJS Instance
        const ctx = document.getElementById('yieldChart').getContext('2d');
        const yieldChart = new Chart(ctx, {
            type: 'line',
            data: {
                labels: xValues,
                datasets: [{
                    label: 'FFB Yield Curve f(x)',
                    data: yValues,
                    borderColor: '#059669', // Emerald 600
                    borderWidth: 3,
                    backgroundColor: 'rgba(5, 150, 105, 0.05)',
                    fill: true,
                    tension: 0.2,
                    pointRadius: 4,
                    pointHoverRadius: 8,
                    pointHoverBackgroundColor: '#d97706', // Amber 600
                    pointHoverBorderColor: '#fff'
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: {
                    legend: { display: false }
                },
                scales: {
                    x: {
                        title: {
                            display: true,
                            text: 'Nitrogen Fertilizer Applied (kg/ha/year)',
                            color: '#374151',
                            font: { weight: 'bold', size: 12 }
                        },
                        grid: { color: '#f3f4f6' }
                    },
                    y: {
                        title: {
                            display: true,
                            text: 'Fresh Fruit Bunch (FFB) Yield (t/ha/year)',
                            color: '#374151',
                            font: { weight: 'bold', size: 12 }
                        },
                        min: 10,
                        max: 32,
                        grid: { color: '#f3f4f6' }
                    }
                },
                // Trigger action on graph interaction
                onClick: (e) => {
                    const points = yieldChart.getElementsAtEventForMode(e, 'nearest', { intersect: false }, true);
                    if (points.length) {
                        const firstPoint = points[0];
                        const index = firstPoint.index;
                        const selectedX = yieldChart.data.labels[index];
                        updateAnalysis(selectedX);
                    }
                }
            }
        });

        // 4. Update the Dashboard Output View 
        function updateAnalysis(x) {
            const fx = getYield(x);
            const dfx = getDerivative(x);

            // Populate Numerical values
            document.getElementById('output-x').innerText = x;
            document.getElementById('output-fx').innerText = fx.toFixed(2);
            document.getElementById('output-dfx').innerText = (dfx >= 0 ? '+' : '') + dfx.toFixed(4);

            // UI Elements to adapt styles based on calculus results
            const statusPill = document.getElementById('status-pill');
            const displayPanel = document.getElementById('analysis-display');
            const explanationText = document.getElementById('output-explanation');

            // 5. Layman Translations Based on Derivative Values
            let statusHTML = '';
            let explanationHTML = '';

            if (dfx > 0.05) {
                // High Positive Derivative: Yield increasing rapidly
                statusPill.className = "px-3 py-1 rounded-full text-xs font-bold bg-emerald-100 text-emerald-800 transition-all duration-300";
                statusPill.innerText = "Status: Efficient Growth Zone";
                explanationHTML = `Adding fertilizer here gives strong results. Every additional 1 kg of Nitrogen increases your oil palm yield by <strong>${dfx.toFixed(3)} tonnes</strong> per hectare. This is a profitable zone to operate in.`;
            } 
            else if (dfx <= 0.05 && dfx > 0) {
                // Low Positive Derivative: Approaching Maximum Point
                statusPill.className = "px-3 py-1 rounded-full text-xs font-bold bg-amber-100 text-amber-800 transition-all duration-300";
                statusPill.innerText = "Status: Near-Optimal Efficiency";
                explanationHTML = `Yield is still increasing, but at a slowing rate. You are approaching the absolute peak crop efficiency. Think carefully: the cost of extra fertilizer might soon outweigh the minor yield gains.`;
            } 
            else if (Math.abs(dfx) <= 0.0001 || (x === 150)) {
                // Derivative is Zero: Peak Yield
                statusPill.className = "px-3 py-1 rounded-full text-xs font-bold bg-blue-100 text-blue-800 transition-all duration-300";
                statusPill.innerText = "Status: Maximum Yield Reached [f'(x) = 0]";
                explanationHTML = `<strong>Perfect balance achieved!</strong> At exactly 150 kg/ha, the derivative is zero, meaning you have maximized your crop potential at <strong>30.00 tonnes/ha</strong>. Any more fertilizer added beyond this point is wasted capital.`;
            } 
            else {
                // Negative Derivative: Over-fertilization / Toxicity
                statusPill.className = "px-3 py-1 rounded-full text-xs font-bold bg-rose-100 text-rose-800 transition-all duration-300";
                statusPill.innerText = "Status: Over-fertilization / Toxic Zone";
                explanationHTML = `<strong>Warning! Toxic Zone.</strong> The rate of change is negative (<strong>${dfx.toFixed(3)}</strong>). Adding more Nitrogen causes root stress and environmental damage, actually shrinking your total FFB harvest yield. Pull back inputs immediately!`;
            }

            explanationText.innerHTML = explanationHTML;
        }
    </script>
